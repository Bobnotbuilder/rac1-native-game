cmake_minimum_required(VERSION 3.20)
# Intentionally separate from the desktop OpenGL build. This is a runtime test target, NOT an app.
project(rac1-apple-runtime LANGUAGES CXX OBJCXX ASM)
if(NOT APPLE OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
    message(FATAL_ERROR "This target requires Apple arm64, not arm64e or an Intel gameplay build")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
include(CheckCXXSourceCompiles)
set(_saved_required_flags "${CMAKE_REQUIRED_FLAGS}")
set(_saved_try_type "${CMAKE_TRY_COMPILE_TARGET_TYPE}")
set(CMAKE_REQUIRED_FLAGS "${CMAKE_REQUIRED_FLAGS} -Werror -frounding-math -ffp-exception-behavior=strict -ffp-contract=off")
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
check_cxx_source_compiles("float add(float a,float b) { return a+b; }" RAC1_COMPILER_HAS_STRICT_ARM64_FP)
set(CMAKE_REQUIRED_FLAGS "${_saved_required_flags}")
if(_saved_try_type)
    set(CMAKE_TRY_COMPILE_TARGET_TYPE "${_saved_try_type}")
else()
    unset(CMAKE_TRY_COMPILE_TARGET_TYPE)
endif()
if(NOT RAC1_COMPILER_HAS_STRICT_ARM64_FP)
    message(FATAL_ERROR "Selected ARM64 compiler lacks strict dynamic floating-point support. Use a validated modern toolchain; do not suppress the diagnostic. Clang 13 experiments are investigation-only, not the production default.")
endif()
include(FetchContent)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
# HostState uses GLFW's public numeric input vocabulary, but never calls its window/input API.
# Fetch just the existing pinned headers; no GLFW/OpenGL library enters the iOS target.
FetchContent_Declare(glfw_input_headers
    URL https://github.com/glfw/glfw/archive/refs/tags/3.4.tar.gz
    URL_HASH SHA256=c038d34200234d071fae9345bc455e4a8f2f544ab60150765d7704e08f3dac01
    SOURCE_SUBDIR _no_cmake)
FetchContent_MakeAvailable(glfw_input_headers)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX,OBJCXX>:-Wall;-Wextra;-Werror;-frounding-math;-ffp-contract=off;-fno-strict-aliasing;-UNDEBUG>")
add_library(rac1_apple_runtime STATIC
    ${ROOT}/src/platform/apple/game_data.cpp
    ${ROOT}/src/platform/apple/platform_apple.mm
    ${ROOT}/src/platform/apple/coroutine_apple.cpp
    ${ROOT}/src/platform/apple/arm64_context.S
    ${ROOT}/src/platform/apple/game_controller.mm
    ${ROOT}/src/host/host_input.cpp
    ${ROOT}/runtime/ee/ee_thread.cpp)
target_include_directories(rac1_apple_runtime PUBLIC ${ROOT}/ios/simd ${ROOT} ${ROOT}/src)
target_include_directories(rac1_apple_runtime SYSTEM PRIVATE ${glfw_input_headers_SOURCE_DIR}/include)
target_compile_definitions(rac1_apple_runtime PRIVATE GLFW_INCLUDE_NONE)
target_compile_options(rac1_apple_runtime PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
target_link_libraries(rac1_apple_runtime PUBLIC "-framework Foundation" "-framework GameController")
add_library(rac1_metal_state STATIC ${ROOT}/src/game/render/metal/gs_blend_metal.mm)
target_include_directories(rac1_metal_state PUBLIC ${ROOT}/src)
target_compile_options(rac1_metal_state PRIVATE -fobjc-arc)
target_link_libraries(rac1_metal_state PUBLIC "-framework Foundation" "-framework Metal")
add_library(rac1_metal_texture STATIC ${ROOT}/src/game/render/metal/mrp_texture.cpp
    ${ROOT}/src/game/render/metal/mrp_texture_metal.mm)
target_include_directories(rac1_metal_texture PUBLIC ${ROOT}/src)
target_compile_options(rac1_metal_texture PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
target_link_libraries(rac1_metal_texture PUBLIC "-framework Foundation" "-framework Metal")
add_library(rac1_metal_scene STATIC ${ROOT}/src/game/render/metal/gs_scene.cpp
    ${ROOT}/src/game/render/metal/gs_scene_metal.mm)
target_include_directories(rac1_metal_scene PUBLIC ${ROOT} ${ROOT}/src)
target_compile_options(rac1_metal_scene PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
target_link_libraries(rac1_metal_scene PUBLIC "-framework Foundation" "-framework Metal")
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    # Same pinned GLM as the original game; headers only for the actual game-host adapter.
    FetchContent_Declare(glm_ios_headers
        URL https://github.com/g-truc/glm/archive/refs/tags/1.0.1.tar.gz
        URL_HASH SHA256=9f3174561fd26904b23f0db5e560971cbf9b3cbda0b280f04d5c379d03bf234c
        SOURCE_SUBDIR _no_cmake)
    FetchContent_MakeAvailable(glm_ios_headers)
    add_library(rac1_ios_host_compile OBJECT ${ROOT}/ios/app/main.mm ${ROOT}/ios/app/game_session.cpp)
    target_link_libraries(rac1_ios_host_compile PRIVATE rac1_apple_runtime rac1_metal_scene)
    target_include_directories(rac1_ios_host_compile SYSTEM PRIVATE ${glm_ios_headers_SOURCE_DIR})
    target_compile_options(rac1_ios_host_compile PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
    target_link_libraries(rac1_ios_host_compile PRIVATE "-framework UIKit" "-framework MetalKit" "-framework AVFAudio")
    # Same pinned miniaudio as the desktop build, without fetching GLFW/OpenGL.
    # Compile the REAL SPU2/IOP output adapter, not a replacement/silent-success audio backend.
    include(FetchContent)
    if(POLICY CMP0135)
        cmake_policy(SET CMP0135 NEW)
    endif()
    FetchContent_Declare(miniaudio
        URL https://github.com/mackron/miniaudio/archive/refs/tags/0.11.25.tar.gz
        URL_HASH SHA256=b900edcffe979816e2560a0580b9b1216d674b4f17fbadeca8f777a7f8ab0274
        SOURCE_SUBDIR _no_cmake)
    FetchContent_MakeAvailable(miniaudio)
    # miniaudio's iOS CoreAudio backend contains AVAudioSession Objective-C code.
    set_source_files_properties(${ROOT}/src/host/audio_out.cpp PROPERTIES LANGUAGE OBJCXX)
    add_library(rac1_ios_audio_compile OBJECT ${ROOT}/src/host/audio_out.cpp)
    target_include_directories(rac1_ios_audio_compile PRIVATE ${ROOT} ${ROOT}/src)
    target_include_directories(rac1_ios_audio_compile SYSTEM PRIVATE ${miniaudio_SOURCE_DIR})
    target_link_libraries(rac1_ios_audio_compile PRIVATE "-framework AVFoundation" "-framework AudioToolbox" "-framework CoreAudio")
endif()
add_library(rac1_vu0_device_contract OBJECT ${ROOT}/tests/apple/vu0_simd_test.cpp)
target_link_libraries(rac1_vu0_device_contract PRIVATE rac1_apple_runtime)
# Compile the actual native VU0 driver too: a SIMD-only fixture did not exercise
# its optional profiler's x86-only intrinsic. No generated game inputs are needed.
add_library(rac1_vu0_driver_compile OBJECT ${ROOT}/runtime/vu/vu0.cpp)
target_link_libraries(rac1_vu0_driver_compile PRIVATE rac1_apple_runtime)
