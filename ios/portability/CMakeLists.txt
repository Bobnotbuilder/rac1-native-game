cmake_minimum_required(VERSION 3.20)
# Intentionally separate from the desktop OpenGL build. This is a runtime test target, NOT an app.
project(rac1-apple-runtime LANGUAGES CXX OBJCXX ASM)
if(NOT APPLE OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
    message(FATAL_ERROR "This target requires Apple arm64, not arm64e or an Intel gameplay build")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
add_compile_options("$<$<COMPILE_LANGUAGE:CXX,OBJCXX>:-Wall;-Wextra;-Werror;-frounding-math;-ffp-contract=off;-fno-strict-aliasing;-UNDEBUG>")
add_library(rac1_apple_runtime STATIC
    ${ROOT}/src/platform/apple/platform_apple.mm
    ${ROOT}/src/platform/apple/coroutine_apple.cpp
    ${ROOT}/src/platform/apple/arm64_context.S
    ${ROOT}/src/platform/apple/game_controller.mm
    ${ROOT}/runtime/ee/ee_thread.cpp)
target_include_directories(rac1_apple_runtime PUBLIC ${ROOT}/ios/simd ${ROOT} ${ROOT}/src)
target_compile_options(rac1_apple_runtime PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
target_link_libraries(rac1_apple_runtime PUBLIC "-framework Foundation" "-framework GameController")
add_library(rac1_metal_state STATIC ${ROOT}/src/game/render/metal/gs_blend_metal.mm)
target_include_directories(rac1_metal_state PUBLIC ${ROOT}/src)
target_compile_options(rac1_metal_state PRIVATE -fobjc-arc)
target_link_libraries(rac1_metal_state PUBLIC "-framework Foundation" "-framework Metal")
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    # Same pinned miniaudio as the desktop build, without fetching GLFW/OpenGL.
    # Compile the REAL SPU2/IOP output adapter, not a replacement/silent-success audio backend.
    include(FetchContent)
    if(POLICY CMP0135)
        cmake_policy(SET CMP0135 NEW)
    endif()
    FetchContent_Declare(miniaudio
        URL https://github.com/mackron/miniaudio/archive/refs/tags/0.11.25.tar.gz
        URL_HASH SHA256=b900edcffe979816e2560a0580b9b1216d674b4f17fbadeca8f777a7f8ab0274
        SOURCE_SUBDIR _no_cmake)
    FetchContent_MakeAvailable(miniaudio)
    # miniaudio's iOS CoreAudio backend contains AVAudioSession Objective-C code.
    set_source_files_properties(${ROOT}/src/host/audio_out.cpp PROPERTIES LANGUAGE OBJCXX)
    add_library(rac1_ios_audio_compile OBJECT ${ROOT}/src/host/audio_out.cpp)
    target_include_directories(rac1_ios_audio_compile PRIVATE ${ROOT} ${ROOT}/src)
    target_include_directories(rac1_ios_audio_compile SYSTEM PRIVATE ${miniaudio_SOURCE_DIR})
    target_link_libraries(rac1_ios_audio_compile PRIVATE "-framework AVFoundation" "-framework AudioToolbox" "-framework CoreAudio")
endif()
add_library(rac1_vu0_device_contract OBJECT ${ROOT}/tests/apple/vu0_simd_test.cpp)
target_link_libraries(rac1_vu0_device_contract PRIVATE rac1_apple_runtime)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    enable_testing()
    add_executable(rac1_apple_runtime_test ${ROOT}/tests/apple/runtime_test.cpp)
    target_link_libraries(rac1_apple_runtime_test PRIVATE rac1_apple_runtime)
    add_test(NAME apple_runtime COMMAND rac1_apple_runtime_test)
    add_executable(rac1_vu0_simd_test ${ROOT}/tests/apple/vu0_simd_test.cpp)
    target_link_libraries(rac1_vu0_simd_test PRIVATE rac1_apple_runtime)
    add_test(NAME vu0_simd COMMAND rac1_vu0_simd_test)
    add_executable(rac1_gs_blend_test ${ROOT}/tests/apple/gs_blend_test.cpp)
    target_link_libraries(rac1_gs_blend_test PRIVATE rac1_metal_state)
    add_test(NAME gs_blend COMMAND rac1_gs_blend_test)
    add_executable(rac1_metal_blend_contract ${ROOT}/tests/apple/metal_blend_contract.mm)
    target_compile_options(rac1_metal_blend_contract PRIVATE -fobjc-arc)
    target_link_libraries(rac1_metal_blend_contract PRIVATE rac1_metal_state)
    add_executable(rac1_metal_contract ${ROOT}/tests/apple/metal_contract.mm)
    target_compile_options(rac1_metal_contract PRIVATE -fobjc-arc)
    target_link_libraries(rac1_metal_contract PRIVATE "-framework Foundation" "-framework Metal")
    add_executable(rac1_programmable_blend_contract ${ROOT}/tests/apple/programmable_blend_contract.mm)
    target_compile_options(rac1_programmable_blend_contract PRIVATE -fobjc-arc)
    target_link_libraries(rac1_programmable_blend_contract PRIVATE "-framework Foundation" "-framework Metal")
endif()
